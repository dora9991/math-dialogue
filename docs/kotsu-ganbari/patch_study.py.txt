import sys
src, dst = sys.argv[1], sys.argv[2]
s = open(src, encoding='utf-8').read()
def rep(old, new):
    global s
    assert s.count(old) == 1, (s.count(old), old[:90])
    s = s.replace(old, new)

# 1) 「きょう・きのう・1週間（きょうをふくむ7日）」の学習時間を出す関数（ktFm のあとに）
KTFM = 'function ktFm(s){s=Math.max(0,Math.floor(Number(s)||0));if(s<60)return s?"1分未満":"0分";const t=Math.floor(s/60);return t<60?`${t}分`:`${Math.floor(t/60)}時間${t%60}分`}'
rep(KTFM, KTFM + 'function ktWk(d){const o=[0,1,2,3,4,5,6].map(i=>{const t=new Date;t.setDate(t.getDate()-i);return Math.max(0,Number(((d||{})[Mt(t)]||{}).sec)||0)});return{t:o[0],y:o[1],w:o.reduce((a,b)=>a+b,0)}}')

# 2) 今日の問題数（問題画面で使う）
rep('ktRt=null,ktTd=0,', 'ktRt=null,ktTd=0,ktTn=0,')
rep('ktRt!==r&&(ktRt=r,ktEp++),ktTd=Math.round(((s.daily||{})[Mt()]||{}).sec||0);',
    'ktRt!==r&&(ktRt=r,ktEp++),ktTd=Math.round(((s.daily||{})[Mt()]||{}).sec||0),ktTn=Math.max(0,Number(((s.daily||{})[Mt()]||{}).n)||0);')

# 3) 問題を解いている画面：「⏱ 今日 ○分 ・ ○問」

        'e.jsx("div",{style:{fontSize:12,fontWeight:800,lineHeight:1.3,width:66,flexShrink:0,whiteSpace:"nowrap"},children:"⏱ 学習時間"}),'
